f_amsClock(ts) =>
    sameDay = year(ts, tzAMS) == year(timenow, tzAMS) and month(ts, tzAMS) == month(timenow, tzAMS) and dayofmonth(ts, tzAMS) == dayofmonth(timenow, tzAMS)
    tmrTs = timestamp(tzAMS, year(timenow, tzAMS), month(timenow, tzAMS), dayofmonth(timenow, tzAMS) + 1, 12, 0, 0)
    isTmr = year(ts, tzAMS) == year(tmrTs, tzAMS) and month(ts, tzAMS) == month(tmrTs, tzAMS) and dayofmonth(ts, tzAMS) == dayofmonth(tmrTs, tzAMS)
    hhmm = str.tostring(hour(ts, tzAMS), "00") + ":" + str.tostring(minute(ts, tzAMS), "00")
    dw = dayofweek(ts, tzAMS)
    dwAr = dw == dayofweek.sunday ? "الأحد" : dw == dayofweek.monday ? "الاثنين" : dw == dayofweek.tuesday ? "الثلاثاء" : dw == dayofweek.wednesday ? "الأربعاء" : dw == dayofweek.thursday ? "الخميس" : dw == dayofweek.friday ? "الجمعة" : "السبت"
    prefix = sameDay ? "" : isTmr ? "غداً " : dwAr + " " + str.tostring(dayofmonth(ts, tzAMS), "00") + "/" + str.tostring(month(ts, tzAMS), "00") + " "
    prefix + hhmm

f_killLines(arr) =>
    sz = array.size(arr)
    if sz > 0
        for i = 0 to sz - 1
            line.delete(array.get(arr, i))
        array.clear(arr)

f_killLabs(arr) =>
    sz = array.size(arr)
    if sz > 0
        for i = 0 to sz - 1
            label.delete(array.get(arr, i))
        array.clear(arr)

f_killBox(arr) =>
    sz = array.size(arr)
    if sz > 0
